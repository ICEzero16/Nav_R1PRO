cmake_minimum_required(VERSION 3.8)
project(r1pro_nav2_critics)
find_package(ament_cmake REQUIRED)
find_package(dwb_core REQUIRED)
find_package(pluginlib REQUIRED)
add_library(goal_orientation SHARED src/goal_orientation.cpp)
target_compile_features(goal_orientation PUBLIC cxx_std_17)
ament_target_dependencies(goal_orientation dwb_core pluginlib)
pluginlib_export_plugin_description_file(dwb_core critics.xml)
install(TARGETS goal_orientation LIBRARY DESTINATION lib)
ament_package()
